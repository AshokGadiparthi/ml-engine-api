url = "http://localhost:8080/api/eda/outliers/{edaId}?method=IQR"

outliers = response.json()

for feature, info in outliers["features"].items():
    print(f"{feature}: {info['outlierCount']} outliers")

EXAMPLE: Data Quality Gate Before Training

@Service
public class TrainingGateway {
    
    @Autowired
    private EDAEnhancedService edaService;
    
    @Autowired
    private TrainingService trainingService;
    
    public TrainingResult trainIfQualityGood(byte[] data, String datasetId) {
        // Run EDA
        Map<String, Object> quality = edaService.getQualityAssessment(data);
        
        // Check quality score
        Double score = (Double) quality.get("overall_score");
        if (score >= 80) {
            // Quality is good - proceed with training
            return trainingService.train(data, datasetId);
        } else {
            throw new RuntimeException("Data quality too low: " + score);
        }
    }
}

TROUBLESHOOTING
===============================================================================

"FastAPI service is not available"
   Solution: Ensure FastAPI is running on port 8001
   $ python -m uvicorn app.main:app --reload --port 8001

"EDAEnhancedService not found"
   Solution: Ensure component scan includes service package
   Check @ComponentScan annotation in main application class

"Timeout on analysis"
   Solution: Increase timeout in application.properties
   fastapi.timeout=60000

"File upload size exceeded"
   Solution: Increase max file size
   spring.servlet.multipart.max-file-size=100MB

"NPE on analysis results"
   Solution: Ensure FastAPI returns proper response format
   Check FastAPI service logs
